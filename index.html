Python SDK

One endpoint. One stable contract.

# pip install tqnn

from tqnn import TQNNClient

client = TQNNClient(api_key="TQNN_xxxxx")

response = client.run_any(
    data=[1.2, 0.4, 3.3, 0.1],
    mode="TABULAR",
    task="fault_diagnosis"
)

print(response["tqnn_report"])
